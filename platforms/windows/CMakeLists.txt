cmake_minimum_required(VERSION 3.19)
project(YANVY_Windows VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "This builds the real Windows TSF platform; use core/ for Linux.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
file(READ "${ROOT}/version.json" YANVY_VERSION_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/version.json")
string(JSON YANVY_VERSION GET "${YANVY_VERSION_JSON}" version)
string(JSON YANVY_WINDOWS_VERSION GET "${YANVY_VERSION_JSON}" windows_version)
string(REPLACE "." "," YANVY_WINDOWS_RC_VERSION "${YANVY_WINDOWS_VERSION}")
configure_file(version.hpp.in version.hpp @ONLY)
set(YANVY_ICON "${ROOT}/assets/brand/yanvy.ico")
configure_file(settings.manifest settings-ui.manifest @ONLY)
set(YANVY_SETTINGS_MANIFEST "${CMAKE_CURRENT_BINARY_DIR}/settings-ui.manifest")
configure_file(installer/ui.manifest.in installer-ui.manifest @ONLY)
set(YANVY_INSTALLER_MANIFEST "${CMAKE_CURRENT_BINARY_DIR}/installer-ui.manifest")
set(YANVY_TESTING_FILE "${CMAKE_CURRENT_SOURCE_DIR}/TESTING.txt")
configure_file(app.rc.in app.rc @ONLY)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/app.rc" PROPERTIES OBJECT_DEPENDS "${YANVY_ICON};${YANVY_TESTING_FILE};${YANVY_INSTALLER_MANIFEST};${YANVY_SETTINGS_MANIFEST}")
add_subdirectory("${ROOT}/core" core)
add_library(YANVY-TSF SHARED src/tsf.cpp src/candidate.cpp src/candidate_view.cpp yanvy.def "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
set_target_properties(YANVY-TSF PROPERTIES PREFIX "")
add_executable(YANVY-Server WIN32 src/server.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
target_link_libraries(YANVY-Server PRIVATE ime_core)
add_executable(YANVY-Setup src/setup.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
add_executable(YANVY-Settings WIN32 src/settings.cpp src/preferences_ui.cpp src/candidate_view.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
target_compile_definitions(YANVY-Settings PRIVATE YANVY_SETTINGS_UI)
add_executable(YANVY-Uninstall WIN32 src/launcher.cpp src/installer_ui.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
set_target_properties(YANVY-Uninstall PROPERTIES OUTPUT_NAME "Uninstall")
target_compile_definitions(YANVY-Uninstall PRIVATE YANVY_INSTALLER_UI)
set(YANVY_TARGETS YANVY-TSF YANVY-Server YANVY-Setup YANVY-Settings YANVY-Uninstall)
add_executable(YANVY-UI-Render EXCLUDE_FROM_ALL tools/render_candidate.cpp src/candidate_view.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
target_include_directories(YANVY-UI-Render PRIVATE src)
list(APPEND YANVY_TARGETS YANVY-UI-Render)
add_executable(YANVY-Installer-Preview WIN32 EXCLUDE_FROM_ALL src/launcher.cpp src/installer_ui.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
target_compile_definitions(YANVY-Installer-Preview PRIVATE YANVY_INSTALLER_UI YANVY_PREVIEW_ONLY)
list(APPEND YANVY_TARGETS YANVY-Installer-Preview)
add_executable(YANVY-Setup-Fixture EXCLUDE_FROM_ALL tools/setup_fixture.cpp)
list(APPEND YANVY_TARGETS YANVY-Setup-Fixture)
add_executable(YANVY-Preferences-Fixture EXCLUDE_FROM_ALL tools/preferences_fixture.cpp src/preferences_ui.cpp src/candidate_view.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
target_include_directories(YANVY-Preferences-Fixture PRIVATE src)
target_compile_definitions(YANVY-Preferences-Fixture PRIVATE YANVY_PREFERENCES_FIXTURE)
list(APPEND YANVY_TARGETS YANVY-Preferences-Fixture)
if(NSIS_PAYLOAD)
  if(NOT EXISTS "${NSIS_PAYLOAD}")
    message(FATAL_ERROR "NSIS_PAYLOAD must point to the compiled internal installer.")
  endif()
  configure_file(installer/payload.rc.in payload.rc @ONLY)
  add_executable(YANVY-Installer WIN32 src/launcher.cpp src/installer_ui.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc" "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
  target_compile_definitions(YANVY-Installer PRIVATE YANVY_INSTALLER_UI)
  list(APPEND YANVY_TARGETS YANVY-Installer)
endif()
foreach(target ${YANVY_TARGETS})
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${ROOT}/core")
  target_link_libraries(${target} PRIVATE ole32 oleaut32 uuid advapi32 user32 gdi32 shell32 dwmapi comctl32 gdiplus)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
    if(target STREQUAL "YANVY-Uninstall" OR target STREQUAL "YANVY-Installer" OR target STREQUAL "YANVY-Settings" OR target STREQUAL "YANVY-Installer-Preview")
      target_link_options(${target} PRIVATE /MANIFEST:NO)
    endif()
  elseif(MINGW)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-missing-field-initializers -finput-charset=UTF-8 -fexec-charset=UTF-8)
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    if(NOT target STREQUAL "YANVY-TSF")
      target_link_options(${target} PRIVATE -municode)
    endif()
  endif()
endforeach()
