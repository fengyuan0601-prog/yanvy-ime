cmake_minimum_required(VERSION 3.19)
project(YANVY_Windows VERSION 0.1.0 LANGUAGES CXX RC)

  message(FATAL_ERROR "This builds the real Windows TSF platform; use core/ for Linux.")

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
file(READ "${ROOT}/version.json" YANVY_VERSION_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/version.json")
string(JSON YANVY_VERSION GET "${YANVY_VERSION_JSON}" version)
string(JSON YANVY_WINDOWS_VERSION GET "${YANVY_VERSION_JSON}" windows_version)
string(JSON YANVY_CHANNEL GET "${YANVY_VERSION_JSON}" channel)

  set(YANVY_RC_FLAGS "0x0L")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(YANVY_BUILD_ARCH "x64")
else()
  set(YANVY_BUILD_ARCH "x86")
endif()
string(REPLACE "." "," YANVY_WINDOWS_RC_VERSION "${YANVY_WINDOWS_VERSION}")

if(NOT YANVY_GIT_RESULT EQUAL 0)
  set(YANVY_SOURCE_COMMIT "unversioned")
endif()
execute_process(COMMAND git rev-parse --git-path HEAD WORKING_DIRECTORY "${ROOT}" OUTPUT_VARIABLE YANVY_GIT_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git symbolic-ref -q HEAD WORKING_DIRECTORY "${ROOT}" OUTPUT_VARIABLE YANVY_GIT_BRANCH OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(YANVY_GIT_HEAD)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/${YANVY_GIT_HEAD}" "${ROOT}/.git/${YANVY_GIT_BRANCH}")
endif()

set(YANVY_INSTALLER_MANIFEST "${CMAKE_CURRENT_BINARY_DIR}/installer-ui.manifest")

configure_file(app.rc.in app.rc @ONLY)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/app.rc" PROPERTIES OBJECT_DEPENDS "${YANVY_ICON};${YANVY_TESTING_FILE};${YANVY_INSTALLER_MANIFEST};${YANVY_SETTINGS_MANIFEST}")

file(GLOB_RECURSE YANVY_CLEANUP_INPUTS CONFIGURE_DEPENDS "${ROOT}/schemas/*.custom.yaml" "${ROOT}/third_party/*")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/scripts/generate_windows_cleanup.py" "${ROOT}/scripts/stage_data.py" ${YANVY_CLEANUP_INPUTS})
execute_process(COMMAND "${Python3_EXECUTABLE}" "${ROOT}/scripts/generate_windows_cleanup.py" --output "${CMAKE_CURRENT_BINARY_DIR}/user_cleanup_list.hpp" RESULT_VARIABLE YANVY_CLEANUP_RESULT)
if(NOT YANVY_CLEANUP_RESULT EQUAL 0)
  message(FATAL_ERROR "Cannot generate the reviewed ordinary-user cleanup list.")
endif()
add_library(YANVY-TSF SHARED src/tsf.cpp src/candidate.cpp src/candidate_view.cpp yanvy.def "${CMAKE_CURRENT_BINARY_DIR}/app.rc")

target_link_libraries(YANVY-Broker-Fixture PRIVATE ime_core)
add_executable(YANVY-Setup src/setup.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
add_executable(YANVY-Settings WIN32 src/settings.cpp src/preferences_ui.cpp src/candidate_view.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
target_compile_definitions(YANVY-Settings PRIVATE YANVY_SETTINGS_UI)

if(NSIS_PAYLOAD)
  if(NOT EXISTS "${NSIS_PAYLOAD}" OR NOT EXISTS "${NSIS_DATA_PAYLOAD}")
    message(FATAL_ERROR "NSIS_PAYLOAD and NSIS_DATA_PAYLOAD must point to the compiled machine and user-data installers.")
  endif()
  configure_file(installer/payload.rc.in payload.rc @ONLY)
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/payload.rc" PROPERTIES OBJECT_DEPENDS "${NSIS_PAYLOAD};${NSIS_DATA_PAYLOAD}")
  add_executable(YANVY-Installer WIN32 src/launcher.cpp src/installer_ui.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc" "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")

foreach(target ${YANVY_TARGETS})
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${ROOT}/core")
  target_link_libraries(${target} PRIVATE ole32 oleaut32 uuid advapi32 user32 gdi32 shell32 dwmapi comctl32 gdiplus)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
    if(target STREQUAL "YANVY-Uninstall" OR target STREQUAL "YANVY-Installer" OR target STREQUAL "YANVY-Settings" OR target STREQUAL "YANVY-Installer-Preview")
      target_link_options(${target} PRIVATE /MANIFEST:NO)
    endif()
  elseif(MINGW)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-missing-field-initializers -finput-charset=UTF-8 -fexec-charset=UTF-8)
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    if(NOT target STREQUAL "YANVY-TSF" AND NOT target STREQUAL "YANVY-TSF-Host-Fixture")
      target_link_options(${target} PRIVATE -municode)
    endif()
  endif()
endforeach()
