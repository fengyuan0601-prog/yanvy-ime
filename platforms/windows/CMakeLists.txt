cmake_minimum_required(VERSION 3.18)
project(YANVY_Windows VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "This builds the real Windows TSF platform; use core/ for Linux.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(YANVY_ICON "${ROOT}/assets/brand/yanvy.ico")
configure_file(app.rc.in app.rc @ONLY)
add_subdirectory("${ROOT}/core" core)
add_library(YANVY-TSF SHARED src/tsf.cpp src/candidate.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
set_target_properties(YANVY-TSF PROPERTIES PREFIX "")
add_executable(YANVY-Server WIN32 src/server.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
target_link_libraries(YANVY-Server PRIVATE ime_core)
add_executable(YANVY-Setup src/setup.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
add_executable(YANVY-Settings WIN32 src/settings.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
foreach(target YANVY-TSF YANVY-Server YANVY-Setup YANVY-Settings)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(${target} PRIVATE ole32 oleaut32 uuid advapi32 user32 gdi32 shell32)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  elseif(MINGW)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-missing-field-initializers -finput-charset=UTF-8 -fexec-charset=UTF-8)
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    if(NOT target STREQUAL "YANVY-TSF")
      target_link_options(${target} PRIVATE -municode)
    endif()
  endif()
endforeach()
